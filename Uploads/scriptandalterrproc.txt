ALTER PROCEDURE [dbo].[USP_AttachedDocument]                                                                                
 @File_Code nvarchar(100)=NULL                                                                              
AS                                                                                
BEGIN                                                                              
 SET NOCOUNT ON;           
    --SELECT   CONCAT('CreatedFile/',DisplayFile) as url,
	SELECT   DisplayFile as url,
	a.Doc_Code,a.Dep_Code,a.Doc_Auth,a.Doc_Keyword,a.Doc_Desc,a.Doc_Lang,a.Doc_Ref,a.Doc_Section,a.Doc_Title,  
	a.Doc_Type,a.Doc_Upload,a.DisplayFile as Doc_Upload_rest,a.Created_By,a.Update_By,    
    case a.Status_Flag when 0 then 'Open' when 1 then 'Approved' when 2 then 'Pending' when -1 then 'Return' when 3 then 'Dispatched' when 4 then 'Close' end as docstatus,                           
    a.Created_DT,a.Update_DT,a.File_Code,a.fin_year,a.Emp_Code,case a.Active when 1 then 'Active' when '0' then 'Reject' end as ActiveStatus,a.Active, b.DocType_Name,                                                                            
    c.Sec_Name,d.Dep_Name, g.DocPrior_Name,a.Status_Flag,a.Attachment_Dt,h.Purpose_desc,a.purpose_type,a.Row_ID,                          
    (select distinct [Name] from Utility_MUser where LoginName=a.Created_By )as  Emp_Name                          
    from  M_Document as a                                                    
    inner join M_DocType as b  on b.DocType_Code= a.Doc_Type                                                                              
    left join M_Section as c  on c.Sec_Code= a.Doc_Section                                                                              
    left join M_Department as d on d.Dep_Code=c.Dep_Code                              
    left join M_DocPrior  as g on g.DocPrior_Code=a.priority                                                                    
    left join M_PurposeType as h on h.Row_ID=a.purpose_type                                                                    
    where   a.File_Code=@File_Code and a.Active=1 and a.main_doc='Y' order by a.Attachment_Dt desc                              
  END
